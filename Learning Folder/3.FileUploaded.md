Option 3: Images + Documents in Same Form

This is what most portfolio/admin panels use.

Route
router.post(
"/upload",
(req, res, next) => {
upload.fields([
{
name: "images",
maxCount: 10,
},
{
name: "documents",
maxCount: 10,
},
])(req, res, next);
},
uploadController
);

But your current code has two separate multer instances (upload and uploadD), so this won't work directly.

Better Professional Approach

Use a single multer configuration.

const storage = multer.diskStorage({
destination: (req, file, cb) => {
const imageTypes = /jpg|jpeg|png|webp/;

    const ext = path
      .extname(file.originalname)
      .toLowerCase();

    if (imageTypes.test(ext)) {
      cb(null, "public/uploads");
    } else {
      cb(null, "public/docs");
    }

},

filename: (req, file, cb) => {
cb(
null,
Date.now() +
"-" +
file.originalname
);
},
});
File Filter
const fileFilter = (req, file, cb) => {
const allowed =
/jpg|jpeg|png|webp|pdf|doc|docx|txt|xlsx|xls|ppt|pptx/;

const ext = allowed.test(
path.extname(file.originalname)
.toLowerCase()
);

if (ext) {
return cb(null, true);
}

cb(
new Error(
"Invalid File Type"
)
);
};
Upload Middleware
const upload = multer({
storage,
fileFilter,
limits: {
fileSize: 10 _ 1024 _ 1024,
},
});

export default upload;
Route
router.post(
"/upload",
upload.array("files", 20),
uploadController
);
Form
<input
  type="file"
  name="files"
  multiple
/>
Controller
const uploadController = (req, res) => {
console.log(req.files);

req.files.forEach((file) => {
console.log(file.filename);
console.log(file.path);
});

res.send("Success");
};

This single-middleware approach is what I would recommend for a real MERN/contact-management project because it handles images and documents together and scales more cleanly.
